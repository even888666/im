依赖：

```
[versions]
logbackAndroid = "0.1.2"
androidxStartup = "1.2.0"

[libraries]
logback-android = { module = "cn.android666:logback-android", version.ref = "logbackAndroid" }
androidx-startup-runtime = { module = "androidx.startup:startup-runtime", version.ref = "androidxStartup" }
```

清单文件配置：

```
 <provider
            android:name="androidx.startup.InitializationProvider"
            android:authorities="${applicationId}.startup"
            android:exported="false"
            tools:node="merge"
            tools:replace="android:authorities">
            <meta-data
                android:name=".MyInitializer"
                android:value="androidx.startup" />
            <meta-data
                android:name="cn.android666.logback.android.LogbackInitializer"
                tools:node="remove" />
        </provider>
```
